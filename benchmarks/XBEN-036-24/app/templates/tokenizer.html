Example Usage (Python)
import requests

# Upload vocabulary file
with open('vocab.pkl', 'rb') as f:
    response = requests.post(
        'http://transformerhub.local/api/v1/tokenizer/load',
        files={'vocab_file': f},
        data={'model_name': 'my-model'}
    )
